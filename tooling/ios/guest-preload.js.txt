// Installed snapshot comes only from the iOS host, never an env/config override.
if ((await import('node:worker_threads')).isMainThread) {
  const guestSnapshot = JSON.parse(process._linkedBinding('dsh_ios_guest_config').read());
  if (guestSnapshot?.installed === true) {
    const { installGuestOwner } = await import('../ios/guest/bootstrap.mjs');
    const module = { exports: {} };
    process.dlopen(module, guestSnapshot.addon);
    installGuestOwner({ native: module.exports, root: guestSnapshot.root,
      protectedPaths: guestSnapshot.protectedPaths, temporaryRoot: guestSnapshot.temporaryRoot, readMetadata: () => guestSnapshot.metadata });
  }
}
